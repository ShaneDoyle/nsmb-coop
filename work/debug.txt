
extern "C"
void printalivetable()
{
	u8* spawnedTable = rcast<u8*>(0x020CA3CC);
	for (u32 i = 0; i < 256; i++) {
		Log::print("%02X ", spawnedTable[i]);
	}
	Log::print("\n");
}

u32 knownLRs[128];

extern "C"
void printUniqueLR(int lr)
{
    for (int i = 0; i < 128; i++)
	{
        if (knownLRs[i] == lr)
            return;
        if (knownLRs[i] == 0)
		{
            knownLRs[i] = lr;
            Log::print("LR: 0x%08X\n", lr);
            return;
        }
    }
}
